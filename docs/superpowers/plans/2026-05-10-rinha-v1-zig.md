# Rinha de Backend 2026 V1 (Zig) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Submeter um baseline funcional em Zig — `/ready` + `/fraud-score` corretos, índice mmap'd compartilhado entre 2 APIs, HAProxy LB — usando **brute-force int8 quantizado** (não float32). Ainda **lento** (alvo p99 ≤ 50 ms para validar pipeline; otimizações de quantização binária + io_uring ficam para V2). Objetivo do V1 é provar a arquitetura end-to-end e ter algo válido na prévia da rinha que **caiba no orçamento de 350 MB**.

**Architecture:** 3 serviços Docker — `lb` (HAProxy 2.9, RR estrito, unix sockets como na V1 Go) + `api1`/`api2` (Zig 0.14, HTTP via std lib bloqueante, mmap RO do índice via volume Docker compartilhado). Index é construído por uma fase `builder` no startup que lê `references.json.gz` (50 MB compresso), normaliza cada um dos 3M registros para 14 floats em [-1, 1], quantiza para int8 (`round(v * 127)` clamped a [-128, 127]) e escreve um arquivo binário compacto. Brute-force int8 puro: para cada query normaliza → 14 floats → quantiza para int8 → varre todos 3M com L2 int8 → top-5 → fraud_score.

**Por que int8 já em V1, não float32:** o kernel Linux carga páginas mmap'd ao primeiro cgroup que dá fault. Com float32 (168 MB de índice) + 2 APIs em cgroups separados, no pior caso uma API faulta tudo na primeira request e ultrapassa seu limite. Int8 (42 MB) cabe folgadamente em 130 MB/API mesmo sob double-counting. Custo: ~30 LOC extras, e estabelece a infra de quantização que V2 precisa para o rerank.

**Tech Stack:** Zig 0.16.x (instalado via `brew install zig`), HAProxy 2.9, Docker (Alpine base), zlib (via `@cImport`) para gzip, std.posix para mmap, std.http para servidor, std.json para parsing.

**Nota Zig 0.16:** APIs mudaram desde 0.14. Em particular: `addExecutable`/`addTest` usam `.root_module = b.createModule(.{ .root_source_file = ..., .target, .optimize, .link_libc })` ao invés do shape antigo com `.root_source_file` direto. `std.io.bufferedReader(...)` foi removido — usar `std.io.bufReader` ou `std.fs.File.reader().interface()`. `std.compress.gzip.decompressor(reader)` mudou de assinatura; verifique `zig build test` cada módulo conforme implementa. Quando em dúvida sobre uma API, consulte context7 com `/websites/ziglang_0_16_0`.

```zig
pub fn build(b: *std.Build) void {
    const target = b.standardTargetOptions(.{});
    const optimize = b.standardOptimizeOption(.{});

    const api = b.addExecutable(.{
        .name = "api",
        .root_source_file = b.path("cmd/api/main.zig"),
        .target = target,
        .optimize = optimize,
    });
    api.linkLibC();
    b.installArtifact(api);

    const builder = b.addExecutable(.{
        .name = "builder",
        .root_source_file = b.path("cmd/builder/main.zig"),
        .target = target,
        .optimize = optimize,
    });
    builder.linkLibC();
    b.installArtifact(builder);

    const run_api = b.addRunArtifact(api);
    if (b.args) |args| run_api.addArgs(args);
    b.step("run-api", "Run the API binary").dependOn(&run_api.step);

    const run_builder = b.addRunArtifact(builder);
    if (b.args) |args| run_builder.addArgs(args);
    b.step("run-builder", "Run the builder binary").dependOn(&run_builder.step);

    const test_step = b.step("test", "Run unit tests");

            .optimize = optimize,
```

```zig
pub fn parseRFC3339Z(s: []const u8) ParseError!i64 {
    if (s.len != 20) return ParseError.InvalidFormat;
    if (s[4] != '-' or s[7] != '-' or s[10] != 'T' or s[13] != ':' or s[16] != ':' or s[19] != 'Z') {
        return ParseError.InvalidFormat;
    }
    const y: i64 = try parseN(s[0..4]);
    const mo: i64 = try parseN(s[5..7]);
    const d: i64 = try parseN(s[8..10]);
    const h: i64 = try parseN(s[11..13]);
    const mi: i64 = try parseN(s[14..16]);
    const se: i64 = try parseN(s[17..19]);

    // Days from 1970-01-01 to (y, mo, d).
    const days = daysFromCivil(y, @intCast(mo), @intCast(d));
    return days * 86400 + h * 3600 + mi * 60 + se;
}

fn parseN(buf: []const u8) ParseError!i64 {
    var v: i64 = 0;
    for (buf) |c| {
        if (c < '0' or c > '9') return ParseError.InvalidFormat;
        v = v * 10 + (c - '0');
    }
    return v;
}

// Howard Hinnant's date algorithm: days since 1970-01-01 for civil (y, m, d).
fn daysFromCivil(y_in: i64, m: u32, d: u32) i64 {
    var y = y_in;
    if (m <= 2) y -= 1;
    const era = if (y >= 0) @divFloor(y, 400) else @divFloor(y - 399, 400);
    const yoe: u32 = @intCast(y - era * 400);
    const m_adj: u32 = if (m > 2) m - 3 else m + 9;
    const doy: u32 = (153 * m_adj + 2) / 5 + d - 1;
    const doe: u32 = yoe * 365 + yoe / 4 - yoe / 100 + doy;
    return era * 146097 + @as(i64, doe) - 719468;
}

pub fn hourUTC(unix_seconds: i64) u8 {
    const sec_of_day = @mod(unix_seconds, 86400);
    return @intCast(@divTrunc(sec_of_day, 3600));
}

pub fn rinhaWeekday(unix_seconds: i64) u8 {
    // 1970-01-01 is Thursday (Go weekday 4). Compute Go weekday then remap.
    const days = @divFloor(unix_seconds, 86400);
    const go_wd: u8 = @intCast(@mod(days + 4, 7));
    return @intCast(@mod(@as(u32, go_wd) + 6, 7));
}
```

## Task 7: Index file format — header + write/read helpers

**Files:**
- Create: `src/index_format.zig`
- Test: inline (round-trip)

V1 format (subject to grow in V2):
```
[header u64 magic] [u64 num_vectors] [u64 dim] [u64 reserved]
[labels: ceil(N/8) bytes]   // 1 = fraud, 0 = legit
[vectors: N * dim * 1 byte (i8, scale = 127, bias = 0)]
```

Quantization scheme: each f32 dim is in [-1, 1] (values clamped via `clamp01` plus the -1 sentinel for missing `last_transaction`). Encoding is `i8 = clamp(round(v * 127), -127, 127)`. Decoding (when needed for inspection) is `f32 = i8 / 127`. This keeps int8 distance proportional to the original L2 distance up to a constant factor — top-K rankings are preserved.

```zig
pub const Header = extern struct {
    magic: u64,
    num_vectors: u64,
    dim: u64,
    reserved: u64,

pub fn labelByteCount(n: u64) u64 {
    return (n + 7) / 8;
}

pub fn vectorByteCount(n: u64, d: u64) u64 {
    return n * d * @sizeOf(i8);
}

pub fn fileSize(n: u64, d: u64) u64 {
    return @sizeOf(Header) + labelByteCount(n) + vectorByteCount(n, d);
}

pub const quant_scale: f32 = 127.0;

pub fn quantize(v: f32) i8 {
    const scaled = v * quant_scale;
    const r = @round(scaled);
    if (r > 127) return 127;
    if (r < -127) return -127;
    return @intFromFloat(r);
}

pub fn quantize14(src: *const [14]f32, dst: *[14]i8) void {
    var i: usize = 0;
    while (i < 14) : (i += 1) dst[i] = quantize(src[i]);
}
```

```zig
pub const Writer = struct {
    file: std.fs.File,
    header: Header,
    label_offset: u64,
    vector_offset: u64,
    written_vectors: u64 = 0,

    pub fn init(file: std.fs.File, n: u64, d: u64) !Writer {
        const h = Header{ .magic = magic, .num_vectors = n, .dim = d, .reserved = 0 };
        try file.writeAll(std.mem.asBytes(&h));
        // Reserve space for labels (zero-filled).
        const label_bytes = labelByteCount(n);
        try file.seekBy(@intCast(label_bytes));
        return .{
            .file = file,
            .header = h,
            .label_offset = @sizeOf(Header),
            .vector_offset = @sizeOf(Header) + label_bytes,
        };
    }

    pub fn writeVector(self: *Writer, vec: []const i8) !void {
        std.debug.assert(vec.len == self.header.dim);
        try self.file.writeAll(std.mem.sliceAsBytes(vec));
        self.written_vectors += 1;
    }

    pub fn setLabel(self: *Writer, idx: u64, fraud: bool) !void {
        const byte_idx = idx / 8;
        const bit_idx: u3 = @intCast(idx % 8);
        try self.file.seekTo(self.label_offset + byte_idx);
        var b: [1]u8 = undefined;
        const n = try self.file.read(&b);
        std.debug.assert(n == 1);
        if (fraud) b[0] |= (@as(u8, 1) << bit_idx) else b[0] &= ~(@as(u8, 1) << bit_idx);
        try self.file.seekTo(self.label_offset + byte_idx);
        try self.file.writeAll(&b);
    }

    pub fn finalize(self: *Writer) !void {
        std.debug.assert(self.written_vectors == self.header.num_vectors);
        try self.file.sync();
    }
};

pub const Reader = struct {
    bytes: []align(@alignOf(Header)) const u8,
    header: Header,
    labels: []const u8,
    vectors: []const i8,

    pub fn init(bytes: []align(@alignOf(Header)) const u8) !Reader {
        if (bytes.len < @sizeOf(Header)) return error.TooShort;
        const h = std.mem.bytesToValue(Header, bytes[0..@sizeOf(Header)]);
        if (h.magic != magic) return error.BadMagic;
        const lb = labelByteCount(h.num_vectors);
        if (bytes.len < @sizeOf(Header) + lb + vectorByteCount(h.num_vectors, h.dim)) return error.TooShort;
        const labels = bytes[@sizeOf(Header) .. @sizeOf(Header) + lb];
        const v_start = @sizeOf(Header) + lb;
        const v_bytes = bytes[v_start..];
        const v_int8 = std.mem.bytesAsSlice(i8, v_bytes);
        return .{ .bytes = bytes, .header = h, .labels = labels, .vectors = v_int8 };
    }

    pub fn vectorAt(self: *const Reader, i: u64) []const i8 {
        const start: u64 = i * self.header.dim;
        return self.vectors[start..][0..self.header.dim];
    }

    pub fn labelAt(self: *const Reader, i: u64) bool {
        return (self.labels[i / 8] >> @as(u3, @intCast(i % 8))) & 1 == 1;
    }
};
```

```zig
pub fn search(reader: *const index_format.Reader, query: *const [14]i8) [k]u64 {
    var top_dist: [k]u32 = .{ std.math.maxInt(u32), std.math.maxInt(u32), std.math.maxInt(u32), std.math.maxInt(u32), std.math.maxInt(u32) };
    var top_idx: [k]u64 = .{ 0, 0, 0, 0, 0 };

    const n = reader.header.num_vectors;
    var i: u64 = 0;
    while (i < n) : (i += 1) {
        const v = reader.vectorAt(i);
        var d: u32 = 0;
        comptime var j: usize = 0;
        inline while (j < 14) : (j += 1) {
            const diff: i32 = @as(i32, query[j]) - @as(i32, v[j]);
            d += @intCast(diff * diff);
        }
        if (d < top_dist[k - 1]) insertSorted(&top_dist, &top_idx, d, i);
    }
    return top_idx;
}

fn insertSorted(dist: *[k]u32, idx: *[k]u64, d: u32, i: u64) void {
    var pos: usize = k - 1;
    while (pos > 0 and dist[pos - 1] > d) : (pos -= 1) {
        dist[pos] = dist[pos - 1];
        idx[pos] = idx[pos - 1];
    }
    dist[pos] = d;
    idx[pos] = i;
}

pub fn fraudScore(reader: *const index_format.Reader, top: [k]u64) f32 {
    var c: u32 = 0;
    for (top) |i| if (reader.labelAt(i)) {
        c += 1;
    };
    return @as(f32, @floatFromInt(c)) / @as(f32, k);
}
```

## Task 12: API server — std.http blocking on unix socket

**Files:**
- Create: `cmd/api/main.zig`

V1 uses Zig's `std.http.Server` on a **unix socket** (HAProxy connects via `unix@/sockets/api1.sock`). Single-threaded, accept-process-loop. mmap the index at startup.

- [ ] **Step 1: Write `cmd/api/main.zig` skeleton**

```zig
const std = @import("std");
const posix = std.posix;
const builtin = @import("builtin");
const index_format = @import("../../src/index_format.zig");
const http_io = @import("../../src/http_io.zig");

pub fn main() !void {
    var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
    defer arena.deinit();
    const ally = arena.allocator();

    var args = try std.process.argsWithAllocator(ally);
    defer args.deinit();
    _ = args.next();
    const sock_path = args.next() orelse "/sockets/api.sock";
    const index_path = args.next() orelse "/index/index.bin";

    std.log.info("api: socket={s} index={s}", .{ sock_path, index_path });

    const idx_file = try std.fs.cwd().openFile(index_path, .{});
    defer idx_file.close();
    const stat = try idx_file.stat();
    const map = try posix.mmap(
        null,
        stat.size,
        posix.PROT.READ,
        .{ .TYPE = .SHARED },
        idx_file.handle,
        0,
    );
    defer posix.munmap(map);

    // mmap returns []align(page) u8; cast align to Header alignment.
    const aligned: []align(@alignOf(index_format.Header)) const u8 = @alignCast(map[0..stat.size]);
    const reader = try index_format.Reader.init(aligned);
    std.log.info("api: index loaded — {} vectors, {} dims", .{ reader.header.num_vectors, reader.header.dim });

    // Remove stale socket if present.
    std.fs.cwd().deleteFile(sock_path) catch |err| switch (err) {
        error.FileNotFound => {},
        else => return err,
    };

    const addr = try std.net.Address.initUnix(sock_path);
    var server = try addr.listen(.{ .reuse_address = true, .kernel_backlog = 1024 });
    defer server.deinit();

    // Make the socket world-writable so HAProxy in another container can dial it.
    try posix.fchmodat(posix.AT.FDCWD, sock_path, 0o666, 0);

    std.log.info("api: listening on {s}", .{sock_path});

    while (true) {
        const conn = server.accept() catch |err| {
            std.log.warn("accept error: {}", .{err});
            continue;
        };
        defer conn.stream.close();

        handleConn(ally, &reader, conn) catch |err| {
            std.log.warn("conn error: {}", .{err});
        };
    }
}

fn handleConn(
    ally: std.mem.Allocator,
    reader: *const index_format.Reader,
    conn: std.net.Server.Connection,
) !void {
    var buf: [8 * 1024]u8 = undefined;
    var http_server = std.http.Server.init(conn, &buf);

    while (http_server.state == .ready) {
        var req = http_server.receiveHead() catch |err| switch (err) {
            error.HttpConnectionClosing => return,
            else => return err,
        };

        if (std.mem.eql(u8, req.head.target, "/ready")) {
            try req.respond("", .{ .status = .ok });
            continue;
        }
        if (req.head.method == .POST and std.mem.eql(u8, req.head.target, "/fraud-score")) {
            const body_reader = try req.reader();
            var body = std.ArrayList(u8).init(ally);
            defer body.deinit();
            try body_reader.readAllArrayList(&body, 64 * 1024);

            var out: [256]u8 = undefined;
            const resp_body = http_io.handle(ally, reader, body.items, &out) catch |err| {
                std.log.warn("handle error: {}", .{err});
                try req.respond("", .{ .status = .internal_server_error });
                continue;
            };
            try req.respond(resp_body, .{
                .status = .ok,
                .extra_headers = &.{.{ .name = "content-type", .value = "application/json" }},
            });
            continue;
        }
        try req.respond("", .{ .status = .not_found });
    }
}
```
